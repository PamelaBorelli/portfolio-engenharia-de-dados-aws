## Etapa 1

Apresente o ator/atriz com maior número de flmes e a respectiva quantidade. A quantidade de filmes encontra-se na coluna **'Number of Movies'** do arquivo.


def actors_max_movie(dataset):
  # Iniciando as variáveis
  ator_max_movie = ''
  max_movie = 0
  # Lendo o dataset linha por linha e eliminando a primeira
  for linha in dataset[1:]:
    # variaveis temporárias
    ator = linha[0]
    num_filmes = int(linha[2])
    # Encontrando o ator com maior número de filme
    if num_filmes > max_movie:
        ator_max_movie = ator
        max_movie = num_filmes
  return ator_max_movie, max_movie

# Imprimindo os resultados
ator, num_filmes = actors_max_movie(dataset)
print(f'O ator/atriz, {ator}, tem o maior número de filmes: {num_filmes}')
